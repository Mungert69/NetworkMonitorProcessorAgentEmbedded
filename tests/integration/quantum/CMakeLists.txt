cmake_minimum_required(VERSION 3.16)
project(nm_quantum_trial C)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WOLFSSL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WOLFSSL_CRYPT_TESTS OFF CACHE BOOL "" FORCE)
set(WOLFSSL_USER_SETTINGS ON CACHE BOOL "" FORCE)
# Source selection must agree with user_settings.h.
set(WOLFSSL_MLKEM ON CACHE BOOL "" FORCE)
set(WOLFSSL_MLDSA ON CACHE BOOL "" FORCE)
set(WOLFSSL_CURVE25519 ON CACHE BOOL "" FORCE)
set(WOLFSSL_TLS13 ON CACHE BOOL "" FORCE)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/config")
add_subdirectory("${REPO}/third_party/wolfssl" wolfssl)
target_compile_options(wolfssl PRIVATE -fwrapv)
add_executable(quantum-probe probe.c "${REPO}/firmware/main/quantum_tls.c")
target_include_directories(quantum-probe PRIVATE "${REPO}/firmware/main")
target_compile_definitions(quantum-probe PRIVATE WOLFSSL_USER_SETTINGS)
find_package(Threads REQUIRED)
target_link_libraries(quantum-probe PRIVATE wolfssl Threads::Threads)
target_compile_options(quantum-probe PRIVATE -Wall -Wextra -Werror)
target_link_options(quantum-probe PRIVATE -Wl,--gc-sections)
add_executable(quantum-provider-checks provider_checks.c "${REPO}/firmware/main/quantum_tls.c")
target_include_directories(quantum-provider-checks PRIVATE "${REPO}/firmware/main")
target_compile_definitions(quantum-provider-checks PRIVATE WOLFSSL_USER_SETTINGS)
target_compile_options(quantum-provider-checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(quantum-provider-checks PRIVATE wolfssl Threads::Threads)
target_link_options(quantum-provider-checks PRIVATE -Wl,--gc-sections)
add_executable(quantum-allocation-checks allocation_checks.c "${REPO}/firmware/main/quantum_tls.c")
target_include_directories(quantum-allocation-checks PRIVATE "${REPO}/firmware/main")
target_compile_definitions(quantum-allocation-checks PRIVATE WOLFSSL_USER_SETTINGS)
target_compile_options(quantum-allocation-checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(quantum-allocation-checks PRIVATE wolfssl Threads::Threads)
target_link_options(quantum-allocation-checks PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--gc-sections)
if(NM_COMMAND_BENCHMARK_FIXTURES)
    add_executable(command-mldsa-checks ../command_benchmark/test_mldsa.c
        "${REPO}/firmware/main/command_mldsa.c" "${REPO}/third_party/yyjson/src/yyjson.c")
    target_include_directories(command-mldsa-checks PRIVATE "${REPO}/firmware/main"
        "${REPO}/third_party/yyjson/src" "${NM_COMMAND_BENCHMARK_FIXTURES}")
    target_compile_definitions(command-mldsa-checks PRIVATE WOLFSSL_USER_SETTINGS)
    target_compile_options(command-mldsa-checks PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(command-mldsa-checks PRIVATE wolfssl Threads::Threads)
    target_link_options(command-mldsa-checks PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
endif()
