option(NM_TEST_BROTLI_Q0 "Test the firmware's specialised pinned Brotli encoder" OFF)
if(NM_TEST_BROTLI_Q0)
  set(NM_BROTLI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/firmware/managed_components/espressif__brotli/brotli")
  if(NOT EXISTS "${NM_BROTLI_SOURCE}/c/enc/encode.c")
    message(FATAL_ERROR "Build firmware once to fetch pinned Brotli before NM_TEST_BROTLI_Q0")
  endif()
  set(BUILD_SHARED_LIBS OFF)
  set(BROTLI_DISABLE_TESTS ON)
  set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${NM_BROTLI_SOURCE}" pinned-brotli EXCLUDE_FROM_ALL)
  get_target_property(reference_sources brotlienc SOURCES)
  list(TRANSFORM reference_sources PREPEND "${NM_BROTLI_SOURCE}/")
  add_library(nm_brotli_reference STATIC ${reference_sources})
  target_include_directories(nm_brotli_reference PUBLIC "${NM_BROTLI_SOURCE}/c/include")
  target_compile_definitions(nm_brotli_reference PRIVATE BROTLI_ENCODER_CLEANUP_ON_OOM)
  target_link_libraries(nm_brotli_reference PUBLIC brotlicommon m)
  include(cmake/brotli-quality-zero.cmake)
  nm_brotli_quality_zero(brotlienc "${NM_BROTLI_SOURCE}" "${Python3_EXECUTABLE}")
  foreach(profile q0 reference)
    add_executable(test_brotli_${profile} tests/native/test_brotli_profile.c)
    target_compile_options(test_brotli_${profile} PRIVATE -UNDEBUG)
    target_link_options(test_brotli_${profile} PRIVATE -Wl,--gc-sections)
    if(profile STREQUAL "q0")
      target_link_libraries(test_brotli_${profile} PRIVATE brotlienc brotlidec)
      target_compile_definitions(test_brotli_${profile} PRIVATE NM_Q0_ONLY=1)
    else()
      target_link_libraries(test_brotli_${profile} PRIVATE nm_brotli_reference brotlidec)
    endif()
    add_test(NAME native-brotli_${profile} COMMAND test_brotli_${profile})
  endforeach()
  add_test(NAME native-brotli_compare COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/native/compare_brotli.py
    $<TARGET_FILE:test_brotli_q0> $<TARGET_FILE:test_brotli_reference>)
  set(BROTLI_ENC brotlienc)
  set(BROTLI_DEC brotlidec)
  set(BROTLI_COMMON brotlicommon)
  set(BROTLI_INCLUDE_DIR "${NM_BROTLI_SOURCE}/c/include")
else()
  find_library(BROTLI_ENC brotlienc REQUIRED)
  find_library(BROTLI_DEC brotlidec REQUIRED)
  find_library(BROTLI_COMMON brotlicommon REQUIRED)
  find_path(BROTLI_INCLUDE_DIR brotli/encode.h REQUIRED)
endif()
